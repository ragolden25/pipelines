== grafana-app-config
---
# Static configs, dashboards/certs, alerting provisioning, datasource template, nsswitch, grafana.ini.
# FIX: dropped grafana/provisioning/api/grafana_api_config.json — that
# was staging for the v13 API-based provisioning flow (push-api-payloads.yml),
# which this role no longer runs now that dashboards/datasources/alerting
# go back to file-based provisioning.
- name: Copy static configuration files
  ansible.builtin.copy:
    src: "{{ grafana_stack_artifact_dir }}/{{ item.src }}"
    dest: "/var/lib/containers/{{ item.dest }}"
    owner: "{{ item.owner | default('65532') }}"
    group: "{{ item.owner | default('65532') }}"
    mode: '0644'
  loop:
    - { src: 'prometheus/prometheus.yml', dest: 'prometheus/prometheus.yml' }
    - { src: 'idrac/idrac.yml', dest: 'idrac/idrac.yml', owner: '0' }
    - { src: 'nginx/nginx.conf', dest: 'nginx/nginx.conf' }
    - { src: 'nginx/mime.types', dest: 'nginx/mime.types' }
    - { src: 'nginx/options-ssl-nginx.conf', dest: 'nginx/options-ssl-nginx.conf' }
    - { src: 'nginx/ssl-dhparams.pem', dest: 'nginx/ssl-dhparams.pem' }
    - { src: 'nginx/conf.d/grafana.conf', dest: 'nginx/conf.d/grafana.conf' }

# FIX: added default.yml — this is the dashboard-provisioning *provider*
# config (tells Grafana which folder to scan for dashboard JSONs). Without
# it, the dashboard JSONs below get copied onto disk but Grafana never
# picks them up. pre-api-main.yml copied this; the v13/API chain dropped
# it since dashboards were being POSTed via REST instead.
- name: Copy loop variables for dashboards and certs
  ansible.builtin.copy:
    src: "{{ grafana_stack_artifact_dir }}/{{ item.dir }}/{{ item.file }}"
    dest: "/var/lib/containers/{{ item.dir }}/{{ item.file }}"
    owner: '65532'
    group: '65532'
    mode: '0644'
  loop:
    - { dir: 'grafana/provisioning/dashboards', file: 'default.yml' }
    - { dir: 'grafana/provisioning/dashboards', file: 'nuc_idrac_dashboard.json' }
    - { dir: 'grafana/provisioning/dashboards', file: 'nuc_linux_dashboard.json' }
    - { dir: 'grafana/provisioning/dashboards', file: 'nuc_windows_dashboard.json' }
    - { dir: 'nginx/certs', file: 'std1-castore.pem' }
    - { dir: 'nginx/certs', file: 'std1.key' }

# FIX: switched from the .json API-payload versions (meant for
# push-api-payloads.yml's REST alerting import, which this role no
# longer runs) back to the native Grafana provisioning .yml format,
# which Grafana reads directly from provisioning/alerting/ on startup —
# same files pre-api-main.yml staged.
- name: Copy Grafana alerting provisioning files
  ansible.builtin.copy:
    src: "{{ grafana_stack_artifact_dir }}/grafana/provisioning/alerting/{{ item }}"
    dest: "/var/lib/containers/grafana/provisioning/alerting/{{ item }}"
    owner: '65532'
    group: '65532'
    mode: '0644'
  loop:
    - contact-points.yml
    - idrac-alerts.yml
    - linux-alerts.yml
    - notification-policies.yml
    - windows-alerts.yml

# FIX: replaced the datasource.json API-payload copy (for
# push-api-payloads.yml's REST datasource import) with the templated
# provisioning-native version pre-api-main.yml used — Grafana reads
# this directly from provisioning/datasources/ on startup, no API call
# needed. Not hostname-templated per your note; std1 is the only
# target right now.
- name: Render Grafana Prometheus datasource (provisioning)
  ansible.builtin.template:
    src: "{{ grafana_stack_artifact_dir }}/grafana/provisioning/datasources/prometheus.yml.j2"
    dest: "/var/lib/containers/grafana/provisioning/datasources/prometheus.yml"
    owner: '65532'
    group: '65532'
    mode: '0644'

- name: Create grafana.access.log
  ansible.builtin.file:
    path: /var/lib/containers/nginx/log/grafana.access.log
    state: touch
    owner: '65532'
    group: '65532'
    mode: '0644'

- name: Create error.log
  ansible.builtin.file:
    path: /var/lib/containers/nginx/log/error.log
    state: touch
    owner: '65532'
    group: '65532'
    mode: '0644'

- name: Install hardened nsswitch.conf for NGINX DNS resolution
  ansible.builtin.copy:
    dest: /var/lib/containers/nginx/nsswitch.conf
    owner: '65532'
    group: '65532'
    mode: '0644'
    content: |
      hosts: files dns
- name: Render grafana.ini
  ansible.builtin.template:
    src: "{{ grafana_stack_artifact_dir }}/grafana/config/grafana.ini.j2"
    dest: /var/lib/containers/grafana/config/grafana.ini
    mode: '0644'
    owner: '65532'
    group: '65532'
