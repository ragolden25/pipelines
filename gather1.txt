# play_weekly_molecule.sh
#!/bin/bash
# play_weekly_molecule.sh - run the Grafana Molecule scenarios after the weekly image builds.
#
# Save as /opt/ansible/nucleus/playbooks/play_weekly_molecule.sh (owner ansible, chmod 755), then as ansible:
# play_weekly_molecule.sh --install-cron adds the Thursday 04:00 crontab line (idempotent)
# play_weekly_molecule.sh run it now (waits if the weekly builds are still running)
#
# Runs, in order: grafana12 (role config) -> grafana12-image nucleus -> grafana12-image ccop
# Reports each to the pipeline dashboard: emit_event.sh check molecule-<name> success|fail "" 604800
# Also writes $TEXTFILE_DIR/pipeline_weekly_molecule.prom for the Nucleus Pipelines Status dashboard.
# Exit status 0 only if every scenario passed.

set -u
RUN_DIR="${RUN_DIR:-/opt/ansible/molecule}"
LOG="${LOG:-/opt/ansible/logs/weekly-molecule.log}"
EMIT="${EMIT:-/usr/local/bin/emit_event.sh}"
TEXTFILE_DIR="${TEXTFILE_DIR:-/var/lib/node_exporter/textfile}"
CLASSES="${MOL_CLASSES:-nucleus ccop}" # image classes to test
WAIT_MAX="${WAIT_MAX:-10800}" # give up waiting for the builds after 3 h
CRON_LINE='0 4 * * 4 /opt/ansible/nucleus/playbooks/play_weekly_molecule.sh >> /opt/ansible/logs/weekly-molecule-cron.log 2>&1'

if [ "${1:-}" = "--install-cron" ]; then
  if crontab -l 2>/dev/null | grep -qF 'play_weekly_molecule.sh'; then
    echo "crontab already has play_weekly_molecule.sh - left alone"
  else
    crontab -l > "$HOME/crontab.bak-$(date +%Y%m%d-%H%M)" 2>/dev/null
    { crontab -l 2>/dev/null; printf '\n# weekly molecule tests (after the weekly image builds)\n%s\n' "$CRON_LINE"; } | crontab - \
      && echo "added: $CRON_LINE"
  fi
  exit 0
fi

mkdir -p "$(dirname "$LOG")"
exec >>"$LOG" 2>&1
log() { echo "$(date -Iseconds) $*"; }

# One run at a time (a manual run and the cron run must not overlap).
exec 9>/tmp/weekly-molecule.lock
flock -n 9 || { log "another weekly molecule run is active - exiting"; exit 1; }

log "=== weekly molecule start ==="

# Wait for the weekly builds to finish (they own the Docker daemon and the scratch dirs).
waited=0
BUILDS_RE="${BUILDS_RE:-^(/bin/|/usr/bin/)?(ba)?sh( -c)? +/opt/ansible/nucleus/playbooks/play_weekly_builds\.sh}"
while pgrep -f "$BUILDS_RE" >/dev/null 2>&1; do
  [ "$waited" -ge "$WAIT_MAX" ] && { log "weekly builds still running after ${WAIT_MAX}s - giving up"; "$EMIT" check molecule-weekly fail "" 604800 >/dev/null 2>&1 || true; exit 1; }
  [ $((waited % 600)) -eq 0 ] && log "weekly builds still running - waiting"
  sleep 60; waited=$((waited + 60))
done

rc=0
METRICS=""
run() { # run <event-name> <scenario> <class> <command...>
  local name=$1 scen=$2 cls=$3; shift 3
  local t0 t1 ok=0 lbl
  t0=$(date +%s); log "--- $name: start ---"
  if "$@"; then
    ok=1; log "--- $name: PASSED ---"; "$EMIT" check "molecule-$name" success "" 604800 >/dev/null 2>&1 || true
  else
    log "--- $name: FAILED (see /opt/ansible/molecule/logs/) ---"; "$EMIT" check "molecule-$name" fail "" 604800 >/dev/null 2>&1 || true
    rc=1
  fi
  t1=$(date +%s)
  lbl="stack=\"grafana\",test=\"$scen\""
  METRICS="$METRICS$(printf 'pipeline_weekly_molecule_status{%s} %s\npipeline_weekly_molecule_timestamp_seconds{%s} %s\npipeline_weekly_molecule_duration_seconds{%s} %s\n' "$lbl" "$ok" "$lbl" "$t1" "$lbl" "$((t1 - t0))")
"
}

run grafana12 role-config - "$RUN_DIR/run-grafana12.sh"
for c in $CLASSES; do
  run "grafana12-image-$c" "image-$c" - "$RUN_DIR/run-grafana12-image.sh" "$c"
done

# Publish to the dashboard through node_exporter's textfile collector (same place the weekly build/CIS metrics go).
if [ -d "$TEXTFILE_DIR" ] && [ -w "$TEXTFILE_DIR" ]; then
  tmp=$(mktemp "$TEXTFILE_DIR/.pipeline_weekly_molecule.XXXXXX")
  {
    echo '# HELP pipeline_weekly_molecule_status Weekly Molecule test: 1 passed, 0 failed.'
    echo '# TYPE pipeline_weekly_molecule_status gauge'
    echo '# HELP pipeline_weekly_molecule_timestamp_seconds Unix time this Molecule scenario finished.'
    echo '# TYPE pipeline_weekly_molecule_timestamp_seconds gauge'
    echo '# HELP pipeline_weekly_molecule_duration_seconds Run time of this Molecule scenario.'
    echo '# TYPE pipeline_weekly_molecule_duration_seconds gauge'
    printf '%s' "$METRICS" | sort -s -t'{' -k1,1
  } > "$tmp" && chmod 0644 "$tmp" && mv -f "$tmp" "$TEXTFILE_DIR/pipeline_weekly_molecule.prom" \
    && log "metrics written to $TEXTFILE_DIR/pipeline_weekly_molecule.prom"
else
  log "WARNING: $TEXTFILE_DIR missing or not writable by $(id -un) - no metrics written (dashboard events were still sent)"
fi

log "=== weekly molecule done (rc=$rc) ==="
exit $rc

# emit_event.sh
#!/usr/bin/env bash
# =====================================================================
# emit_event.sh -- report a pipeline step's result to Prometheus via the
# node_exporter textfile collector.
#
# Usage:
#   emit_event.sh <type> <target> <success|fail> [version] [interval_seconds]
#
#   type      build | sync | scan | sign | stage | deploy | repo
#   target    what it was about: debian13-go, nucleus-nginx, grafana, ...
#   status    success | fail
#   version   optional: image tag/digest, package version, ...
#   interval  optional: how often this job is expected to run, in seconds
#             (86400 daily, 604800 weekly). Lets the dashboard flag stale jobs.
#
# One file per (type,target), written atomically, so jobs never clobber
# each other. Never fails the calling script: errors here are swallowed.
#
# Typical use inside a pipeline script:
#   trap 'rc=$?; emit_event.sh build debian13-go "$([[ $rc -eq 0 ]] && echo success || echo fail)" "$TAG" 86400' EXIT
# =====================================================================

TEXTFILE_DIR="${TEXTFILE_DIR:-/var/lib/node_exporter/textfile}"

main() {
  local type="${1:-}" target="${2:-}" status="${3:-}" version="${4:-}" interval="${5:-0}"

  if [[ -z "$type" || -z "$target" || ( "$status" != "success" && "$status" != "fail" ) ]]; then
    echo "usage: emit_event.sh <type> <target> <success|fail> [version] [interval_seconds]" >&2
    return 2
  fi
  [[ "$interval" =~ ^[0-9]+$ ]] || interval=0

  # Keep label values and the filename safe.
  clean() { printf '%s' "$1" | tr -c 'A-Za-z0-9._:/+-' '_'; }
  type="$(clean "$type")"; target="$(clean "$target")"; version="$(clean "$version")"
  local host; host="$(hostname -s 2>/dev/null || echo unknown)"
  local labels="type=\"${type}\",target=\"${target}\",host=\"${host}\""

  local fname; fname="pipeline_$(printf '%s_%s' "$type" "$target" | tr -c 'A-Za-z0-9_' '_').prom"
  local out="${TEXTFILE_DIR}/${fname}"
  local now; now="$(date +%s)"
  local value=0; [[ "$status" == "success" ]] && value=1

  # On failure, keep the previous last-success time so the dashboard can
  # show "last good build was N days ago".
  local last_success=0
  if [[ $value -eq 1 ]]; then
    last_success="$now"
  elif [[ -r "$out" ]]; then
    last_success="$(awk '/^pipeline_last_success_timestamp_seconds\{/ {print $NF}' "$out" | head -n1)"
    [[ "$last_success" =~ ^[0-9]+$ ]] || last_success=0
  fi

  mkdir -p "$TEXTFILE_DIR" || return 1
  local tmp; tmp="$(mktemp "${TEXTFILE_DIR}/.${fname}.XXXXXX")" || return 1

  cat > "$tmp" <<EOF
# HELP pipeline_last_status 1 if the last run succeeded, 0 if it failed.
# TYPE pipeline_last_status gauge
pipeline_last_status{${labels}} ${value}
# HELP pipeline_last_run_timestamp_seconds Unix time of the last run (success or fail).
# TYPE pipeline_last_run_timestamp_seconds gauge
pipeline_last_run_timestamp_seconds{${labels}} ${now}
# HELP pipeline_last_success_timestamp_seconds Unix time of the last successful run (0 = never seen).
# TYPE pipeline_last_success_timestamp_seconds gauge
pipeline_last_success_timestamp_seconds{${labels}} ${last_success}
# HELP pipeline_expected_interval_seconds How often this job is expected to run (0 = unspecified).
# TYPE pipeline_expected_interval_seconds gauge
pipeline_expected_interval_seconds{${labels}} ${interval}
# HELP pipeline_info Version/tag/digest handled by the last run.
# TYPE pipeline_info gauge
pipeline_info{${labels},version="${version}"} 1
EOF

  chmod 0644 "$tmp"
  mv -f "$tmp" "$out"
}

# Never break the calling pipeline: report problems on stderr but exit 0,
# except for usage errors.
main "$@"
rc=$?
[[ $rc -eq 2 ]] && exit 2
[[ $rc -ne 0 ]] && echo "emit_event.sh: could not write metrics (rc=$rc)" >&2
exit 0

# run-grafana12-image.sh
#!/bin/bash
# run-grafana12-image.sh [nucleus|ccop] - image test of the weekly Grafana + Prometheus images (default: nucleus)
export MOL_IMAGE_CLASS="${1:-nucleus}"
source /opt/ansible/molecule/venv/bin/activate
export XDG_CACHE_HOME=/opt/ansible/molecule/cache
SCR=/opt/ansible/molecule/scratch/containers
mkdir -p "$SCR"
# the folder is written by containers (uid 65532), so clean it with a container too
clean() { docker run --rm -v "$SCR":/s "${MOL_BASE_IMAGE:-debian:trixie}" sh -c 'rm -rf /s/* /s/.[!.]* 2>/dev/null; true'; }
clean
cd /opt/ansible/nucleus/roles/configure_grafana_stack || exit 1
LOG=/opt/ansible/molecule/logs/grafana12-image-$MOL_IMAGE_CLASS-$(date +%F-%H%M).log
molecule test -s grafana12-image 2>&1 | tee "$LOG"
rc=${PIPESTATUS[0]}
clean
echo "log: $LOG (exit $rc)"
exit $rc
